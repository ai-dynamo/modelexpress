{{- if eq .Values.env.MX_METADATA_BACKEND "kubernetes" }}

IMPORTANT: Helm installs missing CRDs from this chart, but does not update CRDs
that already exist. Before upgrading, or when installing on a cluster with an
older ModelExpress installation, apply the CRDs shipped with this chart:

From a source checkout:

  kubectl apply -f helm/crds/modelexpress-crds.yaml

For a packaged or remote chart, extract the CRDs from the same chart version:

  helm show crds CHART_REFERENCE --version {{ .Chart.Version }} | kubectl apply -f -

Set serviceAccount.rbac.enabled=true so the server can manage ModelMetadata and
ModelCacheEntry resources with the Kubernetes backend.
{{- end }}

connection settings) and re-run `helm upgrade`.

Prometheus only adopts a PodMonitor or PrometheusRule whose labels match its
podMonitorSelector / ruleSelector, and kube-prometheus-stack defaults both to
`release: <its own release name>`. An unmatched resource is ignored silently —

{{- else if and .Values.service (contains "NodePort" .Values.service.type) }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "modelexpress.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "gRPC endpoint: $NODE_IP:$NODE_PORT"
{{- else if and .Values.service (contains "LoadBalancer" .Values.service.type) }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch its status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "modelexpress.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "modelexpress.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo "gRPC endpoint: $SERVICE_IP:{{ .Values.service.port | default 8001 }}"
{{- else if and .Values.service (contains "ClusterIP" .Values.service.type) }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "modelexpress.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Port-forward to access gRPC service locally:"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
  echo "gRPC endpoint: localhost:8080"
{{- end }}
